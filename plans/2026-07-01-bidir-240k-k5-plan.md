**Files:**

- Output: `k5_240/rom_words_70_k5.txt`, `k5_240/golden_k5.mat`, `k5_240/PACKET_K5.txt`

- [ ] **Step 1: Extract the legacy conventions (do NOT guess them)**

```bash
grep -n "reshape\|perm\|bi2de\|de2bi\|left-msb\|right-msb\|CAP" /mnt/onetb/scratch/linkA_iq/approachb_bitcontract.m | head -30
```

- [ ] **Step 2: Write `packet_k5.m`**

```matlab
% packet_k5.m -- K=5 bit contract for the 240k/K5 link.
% Payload stays 2240 bits (= 70 uint32) so frame geometry is unchanged (1133 sym).
out = '/mnt/onetb/scratch/qpsk_variants/k5_240/';
trellis = poly2trellis(5,[35 23]); TB = 25;
msg = 'ADI Hello World';
msgBits = reshape(de2bi(uint8(msg),8,'left-msb').',[],1);       % 120x1
rng(9002,'twister'); pad = randi([0 1],964,1);                   % deterministic PN pad
info = [msgBits; pad];                                           % 1084x1
encIn = [info; zeros(4,1)];                                      % + K-1 tail = 1088
coded = convenc(encIn, trellis);  assert(numel(coded)==2176);
ROWS = 136; COLS = 16;
% >>> interleave: REUSE the approachb_bitcontract.m expression verbatim, ROWS=136 <<<
il = reshape(reshape(coded,ROWS,COLS).',[],1);   % REPLACE with legacy expr from Step 1
filler = ones(64,1);
payload = [il; filler];  assert(numel(payload)==2240);
% >>> ROM packing: REUSE legacy packing expression verbatim <<<
words = uint32(bi2de(reshape(payload,32,[]).','left-msb'));      % REPLACE with legacy expr
assert(numel(words)==70);
% ---- round-trip self-check (host Viterbi) ----
rx = payload(1:2176);
deil = zeros(2176,1); % >>> deint: inverse of the legacy perm, ROWS=136 <<<
tmp = reshape(rx,COLS,ROWS).'; deil = tmp(:);    % REPLACE to be exact inverse
dec = vitdec(deil, trellis, TB, 'term', 'hard');
assert(isequal(dec(1:120), msgBits), 'round-trip FAILED');
% CAP_OUT golden: REUSE legacy packing of first 32 recovered info bits
capOut = uint32(bi2de(dec(1:32).','left-msb'));  % REPLACE with legacy expr
fid=fopen([out 'rom_words_70_k5.txt'],'w'); fprintf(fid,'%u\n',words); fclose(fid);
save([out 'golden_k5.mat'],'words','payload','coded','info','msgBits','capOut','trellis','TB','ROWS','COLS');
fid=fopen([out 'PACKET_K5.txt'],'w');
fprintf(fid,'K=5 [35 23] TB=25  info=1084 (120 msg + 964 pad rng(9002)) tail=4 coded=2176 ROWS=136 COLS=16 filler=64 payload=2240\nCAP_OUT_GOLDEN=0x%08X\n',capOut); fclose(fid);
fprintf('PACKET_K5 OK  CAP_OUT=0x%08X\n', capOut);
```

```bash
#!/bin/bash
# arm_zed.sh -- cal + arm ZedBoard for FDD (Tx 2.40 in-FPGA ROM, Rx 2.45), with readbacks
SH=${SH:-"ssh -o StrictHostKeyChecking=no"}
Z=root@10.0.0.128; P=/sys/bus/iio/devices/iio:device0
$SH $Z "cat /root/cmos_repo.bin > $P/stream_config; cat /root/cmos_repo.json > $P/profile_config; sleep 2;
 echo 1 > $P/out_voltage0_lo_leakage_tracking_en; echo 1 > $P/out_voltage0_quadrature_tracking_en;
 echo 2400000000 > $P/out_altvoltage2_TX1_LO_frequency; echo 0 > $P/out_voltage0_hardwaregain;
 echo 2450000000 > $P/out_altvoltage0_RX1_LO_frequency; echo 1 > $P/in_voltage0_quadrature_tracking_en;
 echo manual > $P/in_voltage0_gain_control_mode; echo 24 > $P/in_voltage0_hardwaregain;
 DRA=/sys/kernel/debug/iio/iio:device6/direct_reg_access; echo '0x418 0x2' > \$DRA; echo '0x458 0x2' > \$DRA;
 busybox devmem 0x43C00000 32 1; sleep 1; busybox devmem 0x43C0011C 32 0; busybox devmem 0x43C00118 32 0;
 busybox devmem 0x43C00114 32 1; busybox devmem 0x43C00110 32 1; sleep 0.3; busybox devmem 0x43C00110 32 0"
# --- readbacks: every one must match or exit 1 ---
R=$($SH $Z "echo \$(cat $P/out_altvoltage2_TX1_LO_frequency) \$(cat $P/out_altvoltage0_RX1_LO_frequency) \
 \$(cat $P/in_voltage0_sampling_frequency) \$(cat $P/out_voltage0_hardwaregain); \
 DRA=/sys/kernel/debug/iio/iio:device6/direct_reg_access; echo 0x418 > \$DRA; cat \$DRA")
echo "readback: $R"
echo "$R" | grep -q "2400000000 2450000000 1920000000 0" && echo "$R" | grep -q "0x2$" \
  && echo ARM_ZED_OK || { echo ARM_ZED_FAIL; exit 1; }
```
